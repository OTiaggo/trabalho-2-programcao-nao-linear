def extrair_semente(matricula: str) -> int:
  """Extrai apenas dígitos numéricos de uma matrícula."""
  if not matricula:
    return 0
  digitos = re.findall(r'\d+', str(matricula))
  return int(''.join(digitos)) if digitos else 0
